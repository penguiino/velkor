# API Reference

Base URL:

http://SERVER_IP:8000

---

# Health Check

GET /health

Response:

{
  "ok": true
}

---

# Start Job

POST /start-job

Body:

{
  "workerId": "worker_1",
  "lat": 36.8,
  "lng": 10.1
}

Response:

{
  "jobId": "abc123"
}

---

# Job Ping

POST /job-ping

Body:

{
  "jobId": "abc123",
  "lat": 36.8,
  "lng": 10.1
}

---

# End Job

POST /end-job

Body:

{
  "jobId": "abc123",
  "lat": 36.8,
  "lng": 10.1
}
